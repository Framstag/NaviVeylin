      fails=$(sed -n 's/.*<testsuite[^>]* failures="\([0-9]*\)".*/\1/p' "$d"/TEST-*.xml | awk '{s+=$1} END {print s+0}')
      errs=$(sed -n 's/.*<testsuite[^>]* errors="\([0-9]*\)".*/\1/p' "$d"/TEST-*.xml | awk '{s+=$1} END {print s+0}')

      echo "$d files=$files tests=$tests failures=$fails errors=$errs sha256:$digest"
    done
