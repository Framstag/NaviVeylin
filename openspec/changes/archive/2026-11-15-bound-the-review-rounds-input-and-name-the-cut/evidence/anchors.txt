review_input_head() { # <cap> <source> <destination> → the lines kept
  local cap="$1" src="$2" dest="$3" lines
  { head -c "$cap" "$src" 2>/dev/null || true; } | sed '$d' > "$dest" || : > "$dest"
  lines=$(wc -l < "$dest" 2>/dev/null | tr -d ' \n')
  case "$lines" in ''|*[!0-9]*) lines=0 ;; esac
  printf '%s\n' "$lines"
}

    "$1" "$2" "$3" "$4" >> "$1"
